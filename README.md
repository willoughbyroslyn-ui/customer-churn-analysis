    "## 6. Support Calls & Payment Delay Impact\n",

    "> High support volume and late payments are classic pre-churn signals."

   "cell_type": "code",
   "execution_count": null,
   "id": "support-calls-sql",
   "metadata": {},
   "outputs": [],
   "source": [
    "by_support = con.execute(\"\"\"\n",
    "    SELECT\n",
    "        CASE\n",
    "            WHEN support_calls = 0 THEN '0 calls'\n",
    "            WHEN support_calls BETWEEN 1 AND 2 THEN '1–2 calls'\n",
    "            WHEN support_calls BETWEEN 3 AND 4 THEN '3–4 calls'\n",
    "            WHEN support_calls BETWEEN 5 AND 6 THEN '5–6 calls'\n",
    "            ELSE '7+ calls'\n",
    "        END                                 AS call_bucket,\n",
    "        COUNT(*)                            AS total,\n",
    "        SUM(Churn)                          AS churned,\n",
    "        ROUND(AVG(Churn) * 100, 2)         AS churn_rate_pct\n",
    "    FROM customers\n",
    "    GROUP BY call_bucket\n",
    "    ORDER BY\n",
    "        CASE call_bucket\n",
    "            WHEN '0 calls'   THEN 1\n",
    "            WHEN '1–2 calls' THEN 2\n",
    "            WHEN '3–4 calls' THEN 3\n",
    "            WHEN '5–6 calls' THEN 4\n",
    "            ELSE 5\n",
    "        END\n",
    "\"\"\").df()\n",
    "\n",
    "by_payment = con.execute(\"\"\"\n",
    "    SELECT\n",
    "        CASE\n",
    "            WHEN payment_delay = 0          THEN 'No delay'\n",
    "            WHEN payment_delay BETWEEN 1  AND 10 THEN '1–10 days'\n",
    "            WHEN payment_delay BETWEEN 11 AND 20 THEN '11–20 days'\n",
    "            WHEN payment_delay BETWEEN 21 AND 30 THEN '21–30 days'\n",
    "            ELSE '30+ days'\n",
    "        END                                 AS delay_bucket,\n",
    "        COUNT(*)                            AS total,\n",
    "        SUM(Churn)                          AS churned,\n",
    "        ROUND(AVG(Churn) * 100, 2)         AS churn_rate_pct\n",
    "    FROM customers\n",
    "    GROUP BY delay_bucket\n",
    "    ORDER BY\n",
    "        CASE delay_bucket\n",
    "            WHEN 'No delay'   THEN 1\n",
    "            WHEN '1–10 days'  THEN 2\n",
    "            WHEN '11–20 days' THEN 3\n",
    "            WHEN '21–30 days' THEN 4\n",
    "            ELSE 5\n",
    "        END\n",
    "\"\"\").df()\n",
    "\n",
    "print('--- Support Calls ---')\n",
